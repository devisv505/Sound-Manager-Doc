 public static object Begin()
 {
  if(EditorApplication.isPlaying)throw new Exception("Expected edit mode before capture setup");
  if(Enumerable.Range(0,UnityEngine.SceneManagement.SceneManager.sceneCount).Any(i=>UnityEngine.SceneManagement.SceneManager.GetSceneAt(i).isDirty))throw new Exception("Unsaved scene work");
  if(SessionState.GetBool("M3.Active",false))throw new Exception("Capture setup already active");
  var w=Game;var group=SizeGroup();var gt=group.GetType();var size=(int)GameType.GetProperty("selectedSizeIndex",Flags).GetValue(w);
  SessionState.SetString("M3.Scene",UnityEngine.SceneManagement.SceneManager.GetActiveScene().path);
  SessionState.SetInt("M3.Size",size);SessionState.SetBool("M3.Max",w.maximized);SessionState.SetBool("M3.Background",Application.runInBackground);
  SessionState.SetString("M3.Selection",Selection.activeObject ? GlobalObjectId.GetGlobalObjectIdSlow(Selection.activeObject).ToString() : "");SessionState.SetString("M3.Focus",EditorWindow.focusedWindow ? EditorWindow.focusedWindow.GetEntityId().ToString():"");
  SessionState.SetString("M3.Windows",string.Join(",",Resources.FindObjectsOfTypeAll<EditorWindow>().Select(x=>x.GetEntityId().ToString())));
  var baseline=new {capturedAt=DateTime.UtcNow.ToString("o"),scene=SessionState.GetString("M3.Scene",""),dirty=false,playing=false,paused=EditorApplication.isPaused,background=Application.runInBackground,sizeIndex=size,customSizeCount=gt.GetMethod("GetCustomCount").Invoke(group,null),gameMaximized=w.maximized,selection=SessionState.GetString("M3.Selection",""),windows=Resources.FindObjectsOfTypeAll<EditorWindow>().Select(x=>new {id=x.GetEntityId().ToString(),type=x.GetType().FullName,title=x.titleContent.text,maximized=x.maximized,position=x.position.ToString()}).ToArray()};
  Directory.CreateDirectory(Root+"/validation/m3");Write(Root+"/validation/m3/session-before.json",baseline);
  int built=(int)gt.GetMethod("GetBuiltinCount").Invoke(group,null),custom=(int)gt.GetMethod("GetCustomCount").Invoke(group,null);
  var a=typeof(Editor).Assembly;var kind=Enum.Parse(a.GetType("UnityEditor.GameViewSizeType"),"FixedResolution");
  var captureSize=Activator.CreateInstance(a.GetType("UnityEditor.GameViewSize"),Flags,null,new object[]{kind,1920,1080,"M3 documentation capture"},null);
  gt.GetMethod("AddCustomSize").Invoke(group,new[]{captureSize});SessionState.SetInt("M3.Custom",custom);GameType.GetProperty("selectedSizeIndex",Flags).SetValue(w,built+custom);
  SessionState.SetBool("M3.Active",true);Application.runInBackground=true;w.Focus();return baseline;
 }
 static void SavePixels(RenderTexture rt,string file){var old=RenderTexture.active;var tex=new Texture2D(rt.width,rt.height,TextureFormat.RGB24,false);try{RenderTexture.active=rt;tex.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);tex.Apply();if(SystemInfo.graphicsUVStartsAtTop){var p=tex.GetPixels32();var row=new Color32[rt.width];for(int y=0;y<rt.height/2;y++){int top=y*rt.width,bottom=(rt.height-1-y)*rt.width;Array.Copy(p,top,row,0,rt.width);Array.Copy(p,bottom,p,top,rt.width);Array.Copy(row,0,p,bottom,rt.width);}tex.SetPixels32(p);tex.Apply();}File.WriteAllBytes(file,tex.EncodeToPNG());}finally{RenderTexture.active=old;UnityEngine.Object.DestroyImmediate(tex);}}
 public static object GameShot(string demo,string name,object state,string[] steps)
 {
  var rt=(RenderTexture)GameType.GetField("m_RenderTexture",Flags).GetValue(Game);
  if(!rt || rt.width!=1920 || rt.height!=1080)throw new Exception("Need rendered 1920x1080 Game View");
  var dir=Root+"/media-source/unity/"+demo;Directory.CreateDirectory(dir);var file=dir+"/"+name+".png";SavePixels(rt,file);
  var hud=UnityEngine.Object.FindObjectsByType<UnityEngine.UIElements.UIDocument>(FindObjectsInactive.Exclude).SelectMany(x=>x.rootVisualElement.Query<TextElement>().ToList()).Where(x=>x.resolvedStyle.display!=DisplayStyle.None).Select(x=>x.text).Where(x=>!string.IsNullOrEmpty(x)).ToArray();
  var meta=new {demo,name,kind="game-view",capturedAt=DateTime.UtcNow.ToString("o"),scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,width=rt.width,height=rt.height,hudIncluded=true,state,steps,hud,unityVersion=Application.unityVersion};Write(dir+"/"+name+".capture.json",meta);return meta;
 }
