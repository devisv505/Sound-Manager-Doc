  var original=v.Q<VisualElement>("content-view-container");
  var key="M3.Graph."+w.GetEntityId();
  if(!SessionState.GetBool(key+".Stored",false)){SessionState.SetVector3(key+".Position",original.resolvedStyle.translate);SessionState.SetVector3(key+".Scale",original.resolvedStyle.scale.value);SessionState.SetBool(key+".Stored",true);}

  var type=v.GetType().Assembly.GetType("Unity.GraphToolkit.Editor.ShortcutFrameAllEvent");var pool=typeof(EventBase<>).MakeGenericType(type).GetMethod("GetPooled",BindingFlags.Public|BindingFlags.Static|BindingFlags.FlattenHierarchy,null,Type.EmptyTypes,null);using(var ev=(EventBase)pool.Invoke(null,null)){v.SendEvent(ev);}await Task.Delay(400);
  return new {name,window=w.GetType().FullName,view=v.GetType().FullName,width=w.position.width,height=w.position.height,nodes=v.Query<VisualElement>(className:"ge-node").ToList().Count};
 }
 static void Save(RenderTexture rt,string file){var old=RenderTexture.active;var tex=new Texture2D(rt.width,rt.height,TextureFormat.RGB24,false);try{RenderTexture.active=rt;tex.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);tex.Apply();if(SystemInfo.graphicsUVStartsAtTop){var p=tex.GetPixels32();var row=new Color32[rt.width];for(int y=0;y<rt.height/2;y++){int top=y*rt.width,bottom=(rt.height-1-y)*rt.width;Array.Copy(p,top,row,0,rt.width);Array.Copy(p,bottom,p,top,rt.width);Array.Copy(row,0,p,bottom,rt.width);}tex.SetPixels32(p);tex.Apply();}File.WriteAllBytes(file,tex.EncodeToPNG());}finally{RenderTexture.active=old;UnityEngine.Object.DestroyImmediate(tex);}}
 public static object Shot(string title,string demo,string name,string asset){return ShotWindow(Resources.FindObjectsOfTypeAll<EditorWindow>().First(w=>w.titleContent.text==title),demo,name,asset,"graph-editor");}
 static object ShotWindow(EditorWindow w,string demo,string name,string asset,string kind){var host=typeof(EditorWindow).GetField("m_Parent",F).GetValue(w);var scale=EditorGUIUtility.pixelsPerPoint;int width=(int)(w.position.width*scale),height=(int)(w.position.height*scale);var rt=new RenderTexture(width,height,0,RenderTextureFormat.ARGB32);rt.Create();var dir=Root+"/media-source/unity/"+demo;Directory.CreateDirectory(dir);try{host.GetType().GetMethod("GrabPixels",F).Invoke(host,new object[]{rt,new Rect(0,0,width,height)});Save(rt,dir+"/"+name+".png");}finally{rt.Release();UnityEngine.Object.DestroyImmediate(rt);}var meta=new {demo,name,kind,asset,capturedAt=DateTime.UtcNow.ToString("o"),width,height,hudIncluded=false,steps=new[]{"Exit Play Mode and open "+asset+" in Unity.","Frame the saved graph or expand the relevant Inspector sections; capture the editor's own GUI surface."},unityVersion=Application.unityVersion};File.WriteAllText(dir+"/"+name+".capture.json",Newtonsoft.Json.JsonConvert.SerializeObject(meta,Newtonsoft.Json.Formatting.Indented));return meta;}
 public static async Task<object> Overview(string path,string demo,string name){await Open(path);await FrameAll(Path.GetFileNameWithoutExtension(path));return Shot(Path.GetFileNameWithoutExtension(path),demo,name,path);}
 public static object Nodes(string title){var w=Resources.FindObjectsOfTypeAll<EditorWindow>().First(w=>w.titleContent.text==title);var v=w.rootVisualElement.Q<VisualElement>(className:"ge-graph-view");var c=v.Q<VisualElement>("content-view-container");return v.Query<VisualElement>().ToList().Where(e=>e.ClassListContains("ge-node")||e.GetType().Name.Contains("Placemat")).Select(e=>new{type=e.GetType().Name,labels=e.Query<TextElement>().ToList().Select(x=>x.text).Where(x=>!string.IsNullOrEmpty(x)).ToArray(),position=c.WorldToLocal(e.worldBound.position).ToString(),size=e.layout.size.ToString(),classes=string.Join(" ",e.GetClasses())}).ToArray();}
 public static async Task<object> Region(string title,float x,float y,float width,float height,float zoom){var w=Resources.FindObjectsOfTypeAll<EditorWindow>().First(w=>w.titleContent.text==title);var v=w.rootVisualElement.Q<VisualElement>(className:"ge-graph-view");var scale=Mathf.Min(zoom,Mathf.Min((v.layout.width-110)/width,(v.layout.height-100)/height));var pos=new Vector3((v.layout.width-width*scale)/2-x*scale,(v.layout.height-height*scale)/2-y*scale,0);v.GetType().GetMethod("UpdateViewTransform",F).Invoke(v,new object[]{pos,new Vector3(scale,scale,1)});await Task.Delay(400);return new{pos=pos.ToString(),scale};}
 public static async Task<object> FrameAll(string title){var w=Resources.FindObjectsOfTypeAll<EditorWindow>().First(w=>w.titleContent.text==title);var v=w.rootVisualElement.Q<VisualElement>(className:"ge-graph-view");var c=v.Q<VisualElement>("content-view-container");var nodes=v.Query<VisualElement>().ToList().Where(e=>e.ClassListContains("ge-node")||e.GetType().Name=="Placemat"||e.GetType().Name=="StickyNote").ToArray();if(nodes.Length==0)throw new Exception("No nodes");var rects=nodes.Select(e=>new Rect(c.WorldToLocal(e.worldBound.min),c.WorldToLocal(e.worldBound.max)-c.WorldToLocal(e.worldBound.min))).ToArray();float x=rects.Min(r=>r.xMin)-40,y=rects.Min(r=>r.yMin)-65,right=rects.Max(r=>r.xMax)+40,bottom=rects.Max(r=>r.yMax)+40;return await Region(title,x,y,right-x,bottom-y,1f);}
